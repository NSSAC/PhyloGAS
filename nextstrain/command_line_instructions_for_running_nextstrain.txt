To run nextstrain (adjustments will need to be made depending on the dataset)
- From synthetic_biosurveillance/nextstrain directory, run nextstrain shell . (does not have to be run from that directory, but relevant data and configuration files are in the directory.)
- augur index --sequences data/run_03_with_pid_proportional_all_in_one.sequences.fasta.xz --output results/run_03_with_pid_proportional_all_in_one.sequence_index.tsv
- augur filter --sequences data/run_03_with_pid_proportional_all_in_one.sequences.fasta.xz --sequence-index results/run_03_with_pid_proportional_all_in_one.sequence_index.tsv --metadata data/run_03_with_pid_proportional_all_in_one.metadata.tsv.xz --output results/filtered.fasta --group-by country year month sex county latitude longitude age_group smh_race --subsample-max-sequences 1000 --probabilistic-sampling
- augur align --sequences results/filtered.fasta --reference-sequence data/GCA_009858895.3_ASM985889v3_genomic.fna --output results/aligned.fasta --fill-gaps
- augur tree --alignment results/aligned.fasta --output results/tree_raw.nwk
- augur refine --tree results/tree_raw.nwk --alignment results/aligned.fasta --metadata data/run_03_with_pid_proportional_all_in_one.metadata.tsv.xz --output-tree results/tree.nwk --output-node-data results/branch_lengths.json --timetree --coalescent opt --date-confidence --date-inference marginal --clock-filter-iqd 4 --clock-rate 0.0008 --clock-std-dev 0.0004
- augur traits --tree results/tree.nwk --metadata data/run_03_with_pid_proportional_all_in_one.metadata.tsv.xz --output-node-data results/traits.json --columns region country county --confidence
- augur ancestral --tree results/tree.nwk --alignment results/aligned.fasta --output-node-data results/nt_muts.json --inference joint
- augur translate --tree results/tree.nwk --ancestral-sequences results/nt_muts.json --output-node-data results/aa_muts.json --reference-sequence config/genomic.gbff
- augur export v2 --tree results/tree.nwk --metadata data/run_03_with_pid_proportional_all_in_one.sequences.metadata.tsv.xz --node-data results/branch_lengths.json results/traits.json results/nt_muts.json results/aa_muts.json --auspice-config config/auspice_augmented.json --lat-longs config/va_lat_longs.tsv --output auspice/syncovid.json
- exit
- nextstrain view auspice
